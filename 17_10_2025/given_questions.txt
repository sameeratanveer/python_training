'''

        for element2 in element:

print(outp)

'''
Q3. Group Consecutive Elements
nums = [1, 1, 2, 3, 3, 3, 4, 1, 1]
Expected Output: [[1, 1], [2], [3, 3, 3], [4], [1, 1]]
'''

PROGRAM1: 

nums = [1, 1, 2, 3, 3, 3, 4, 1, 1]
outp = []
i= 0

while i < len(nums):
    j = i
    while j < len(nums) - 1 and nums[j] == nums[j + 1]:
        j += 1
    outp.append(nums[i:j+1])
    i = j + 1

print(outp)

OUTPUT:

[[1, 1], [2], [3, 3, 3], [4], [1, 1]]

PROGRAM 2:

nums = [1, 1, 2, 3, 3, 3, 4, 1, 1]
outp = []
current = [nums[0]]

for i in range(1, len(nums)):
    if nums[i] == nums[i-1]:
        current.append(nums[i])
    else:
        outp.append(current)
        current = [nums[i]]
outp.append(current)
print(outp)

OUTPUT:

[[1, 1], [2], [3, 3, 3], [4], [1, 1]]

'''
Q12. Merge Dicts and Sum Common Keys
d1 = {'a': 10, 'b': 20, 'c': 30}
d2 = {'a': 5, 'b': 15, 'd': 25}
Expected Output: {'a': 15, 'b': 35, 'c': 30, 'd': 25}
'''

PROGRAM :

d1 = {'a': 10, 'b': 20, 'c': 30}
d2 = {'a': 5, 'b': 15, 'd': 25}

outp = d1.copy()
for key, value in d2.items():
    if key not in outp:
        outp[key] = value
    else:
        outp[key] += value
print(outp)

OUTPUT:

{'a': 15, 'b': 35, 'c': 30, 'd': 25}

PROGRAM 2:
dict1 = {'a': 10, 'b': 20, 'c': 30}
dict2 = {'b': 5, 'c': 15, 'd': 25}

outp = {k: dict1.get(k, 0) + dict2.get(k, 0) for k in set(dict1) | set(dict2)}
print(outp)

OUTPUT:

{'a': 15, 'b': 35, 'c': 30, 'd': 25}

'''
Q13. Flatten Nested Dictionary
data = {'a': {'b': {'c': 1}}, 'd': 2}
Expected Output: {'a.b.c': 1, 'd': 2}
'''

PROGRAM

data = {'a': {'b': {'c': 1}}, 'd': 2}
outp = {}
for key, value in data.items():
    if isinstance(value, dict):
        for key2, value2 in value.items():
            if isinstance(value2, dict):
                for key3, value3 in value2.items():
                    keyc = f'{key}.{key2}.{key3}'
                    outp[keyc] = value3
            else:
                keyc = f'{key}.{key2}'
                outp[keyc] = value2
    else:
        outp[key] = value
print(outp)

OUTPUT:
{'a.b.c': 1, 'd': 2}

'''
Q14. Group by Value
data = {'x': 1, 'y': 2, 'z': 1}
Expected Output: {1: ['x', 'z'], 2: ['y']}
'''

PROGRAM:

data = {'x': 1, 'y': 2, 'z': 1}
outp = {}

for key, value in data.items():
    if value not in outp:
        outp[value] = [key]
    else:
        outp[value].append(key)
print(outp)

OUTPUT:
{1: ['x', 'z'], 2: ['y']}

PROGRAM 2:

data = {"a": 1, "b": 2, "c": 1, "d": 3, "e": 2}
outp = {val: [k for k in data if data[k] == val] for val in set(data.values())}
print(outp)

OUTPUT:
{1: ['x', 'z'], 2: ['y']}

data = {'a': {'id': 1}, 'b': {'id': 2}, 'c': {'id': 1}}
